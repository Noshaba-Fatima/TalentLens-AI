from collections import Counter

from app.schemas.assessment import (
    Assessment,
    AssessmentReview,
    ReviewIssue
)
from app.graph.state import TalentLensState


def validate_assessment(state: TalentLensState) -> dict:

    assessment = state["assessment"]
    plan = assessment.assessment_plan

    issues = []

    # --------------------------------------------------
    # 1. VALIDATE TOTAL QUESTION COUNT
    # --------------------------------------------------

    actual_total = len(assessment.questions)

    if actual_total != plan.total_questions:

        issues.append(
            ReviewIssue(
                issue_type="question_count_mismatch",
                description=(
                    f"Expected {plan.total_questions} questions, "
                    f"but generated {actual_total}."
                ),
                suggested_fix=(
                    "Generate the exact number of questions "
                    "specified in the assessment plan."
                )
            )
        )

    # --------------------------------------------------
    # 2. VALIDATE SKILL DISTRIBUTION
    # --------------------------------------------------

    generated_skill_counts = Counter(
        question.skill_name.strip().casefold()
        for question in assessment.questions
    )

    for skill in plan.skill_assessments:

        skill_key = skill.skill_name.strip().casefold()

        expected_count = skill.question_count
        actual_count = generated_skill_counts.get(skill_key, 0)

        if actual_count != expected_count:

            issues.append(
                ReviewIssue(
                    issue_type="skill_distribution_mismatch",
                    description=(
                        f"Skill '{skill.skill_name}': expected "
                        f"{expected_count} questions, but generated "
                        f"{actual_count}."
                    ),
                    suggested_fix=(
                        f"Generate exactly {expected_count} questions "
                        f"for {skill.skill_name}."
                    )
                )
            )

    # --------------------------------------------------
    # 3. VALIDATE DIFFICULTY
    # --------------------------------------------------

    planned_difficulties = {
        skill.skill_name.strip().casefold(): skill.difficulty
        for skill in plan.skill_assessments
    }

    for question in assessment.questions:

        skill_key = question.skill_name.strip().casefold()
        expected_difficulty = planned_difficulties.get(skill_key)

        if (
            expected_difficulty is not None
            and question.difficulty != expected_difficulty
        ):

            issues.append(
                ReviewIssue(
                    issue_type="difficulty_mismatch",
                    question_id=question.question_id,
                    description=(
                        f"Question difficulty for '{question.skill_name}' "
                        f"is {question.difficulty.value}, but the plan "
                        f"requires {expected_difficulty.value}."
                    ),
                    suggested_fix=(
                        f"Adjust the question difficulty to "
                        f"{expected_difficulty.value}."
                    )
                )
            )

    # --------------------------------------------------
    # 4. VALIDATE QUESTION TYPES
    # --------------------------------------------------

    planned_question_types = {
        skill.skill_name.strip().casefold(): set(skill.question_types)
        for skill in plan.skill_assessments
    }

    for question in assessment.questions:

        skill_key = question.skill_name.strip().casefold()
        allowed_types = planned_question_types.get(skill_key)

        if (
            allowed_types is not None
            and question.question_type not in allowed_types
        ):

            issues.append(
                ReviewIssue(
                    issue_type="question_type_mismatch",
                    question_id=question.question_id,
                    description=(
                        f"Question type '{question.question_type.value}' "
                        f"is not allowed for skill '{question.skill_name}'."
                    ),
                    suggested_fix=(
                        "Use one of the question types specified "
                        "in the assessment plan."
                    )
                )
            )

    # --------------------------------------------------
    # 5. VALIDATE QUESTION MARKS
    # --------------------------------------------------

    for question in assessment.questions:

        if question.max_marks <= 0:

            issues.append(
                ReviewIssue(
                    issue_type="invalid_marks",
                    question_id=question.question_id,
                    description="Question has non-positive marks.",
                    suggested_fix="Assign positive marks to the question."
                )
            )

        if not question.evaluation_rubric:

            issues.append(
                ReviewIssue(
                    issue_type="missing_rubric",
                    question_id=question.question_id,
                    description="Question has no evaluation rubric.",
                    suggested_fix="Add measurable evaluation criteria."
                )
            )

    # --------------------------------------------------
    # 6. VALIDATE TIME LIMIT
    # --------------------------------------------------

    total_question_time = sum(
        question.time_limit_minutes
        for question in assessment.questions
    )

    if total_question_time > plan.duration_minutes:

        issues.append(
            ReviewIssue(
                issue_type="time_limit_exceeded",
                description=(
                    f"Total question time is {total_question_time} minutes, "
                    f"but assessment duration is "
                    f"{plan.duration_minutes} minutes."
                ),
                suggested_fix=(
                    "Reduce individual question time limits or "
                    "increase the assessment duration."
                )
            )
        )

    # --------------------------------------------------
    # 7. VALIDATE DUPLICATE QUESTION IDS
    # --------------------------------------------------

    question_ids = [
        question.question_id
        for question in assessment.questions
    ]

    id_counts = Counter(question_ids)

    for question_id, count in id_counts.items():

        if count > 1:

            issues.append(
                ReviewIssue(
                    issue_type="duplicate_question_id",
                    question_id=question_id,
                    description=(
                        f"Question ID '{question_id}' appears "
                        f"{count} times."
                    ),
                    suggested_fix=(
                        "Assign a unique ID to every assessment question."
                    )
                )
            )

    # --------------------------------------------------
    # 8. BUILD REVIEW RESULT
    # --------------------------------------------------

    is_valid = len(issues) == 0

    if is_valid:
        overall_feedback = (
            "Assessment passed all deterministic validation checks."
        )
    else:
        overall_feedback = (
            f"Assessment failed validation with {len(issues)} issue(s)."
        )

    review = AssessmentReview(
        is_valid=is_valid,
        overall_feedback=overall_feedback,
        issues=issues,
        revision_suggestions=[
            issue.suggested_fix for issue in issues
        ]
    )

    return {
        "review_result": review,
        "is_valid": is_valid,
        "review_feedback": overall_feedback
    }

JOB DESCRIPTION:

{job_description_text}

ASSESSMENT PLAN:

{assessment_plan_text}

PREVIOUS ASSESSMENT:

{previous_assessment.model_dump_json(indent=2)}

REVIEW FEEDBACK:

{review_feedback or "No specific review feedback provided."}

Revise the previous assessment.

Preserve correct questions and fix the identified issues.

    else:

        human_prompt += """

This is the initial assessment generation.

Generate a new assessment from the job description
and assessment plan.

Return exactly the number of questions specified
in the assessment plan.
"""

        assessment_plan=assessment_plan,

    )

    return {
        "assessment": assessment

def combine_reviews(state: TalentLensState) -> dict:

    deterministic_review = state["review_result"]
    quality_review = state["quality_review"]

    combined_issues = (
        deterministic_review.issues
        + quality_review.issues
    )

    combined_suggestions = (
        deterministic_review.revision_suggestions
        + quality_review.revision_suggestions
    )

    is_valid = (
        deterministic_review.is_valid
        and quality_review.is_valid
    )

    if is_valid:
        overall_feedback = (
            "Assessment passed deterministic and LLM quality checks."
        )

        revision_feedback = overall_feedback

    else:
        overall_feedback = (
            f"Assessment failed review with "
            f"{len(combined_issues)} issue(s)."
        )

        # Give the question generator actionable feedback
        feedback_parts = []

        for issue in combined_issues:
            feedback_parts.append(
                f"""
                Issue: {issue.issue_type}
                Question ID: {issue.question_id}
                Description: {issue.description}
                Suggested Fix: {issue.suggested_fix}
                """
            )

        revision_feedback = (
            overall_feedback
            + "\n\nDetailed review feedback:\n"
            + "\n".join(feedback_parts)
        )

    final_review = AssessmentReview(
        is_valid=is_valid,
        overall_feedback=overall_feedback,
        issues=combined_issues,
        revision_suggestions=combined_suggestions
    )

    return {
        "final_review": final_review,
        "is_valid": is_valid,
        "review_feedback": revision_feedback
    }

from app.graph.state import TalentLensState

# Maximum number of revisions after the initial generation
MAX_RETRIES = 2


# --------------------------------------------------
# REVISION CONTROL
# --------------------------------------------------

def route_after_review(state: TalentLensState) -> str:
    """
    Decide whether the assessment should be accepted,
    revised, or returned as rejected after retry exhaustion.
    """

    if state["is_valid"]:
        return "end"

    revision_count = state.get("revision_count", 0)

    if revision_count < MAX_RETRIES:
        return "revise"

    return "end"


def prepare_revision(state: TalentLensState) -> dict:
    """
    Increment the revision counter before regenerating
    the assessment.
    """

    current_count = state.get("revision_count", 0)

    return {
        "revision_count": current_count + 1
    }

# BUILD LANGGRAPH WORKFLOW
# --------------------------------------------------

builder = StateGraph(TalentLensState)


# Register nodes
builder.add_node("jd_analyzer", analyze_job_description)
builder.add_node("assessment_planner", plan_assessment)
builder.add_node("question_generator", generate_questions)
builder.add_node("deterministic_validator", validate_assessment)
builder.add_node("quality_reviewer", review_assessment_quality)
builder.add_node("review_combiner", combine_reviews)
builder.add_node("prepare_revision", prepare_revision)


# Initial execution flow
builder.add_edge(START, "jd_analyzer")

builder.add_edge("jd_analyzer", "assessment_planner")
builder.add_edge("assessment_planner", "question_generator")

builder.add_edge("question_generator", "deterministic_validator")
builder.add_edge("deterministic_validator", "quality_reviewer")
builder.add_edge("quality_reviewer", "review_combiner")


# Conditional routing after both reviews are combined
builder.add_conditional_edges(
    "review_combiner",
    route_after_review,
    {
        "revise": "prepare_revision",
        "end": END
    }
)


# Revision loop
builder.add_edge("prepare_revision", "question_generator")


# Compile the graph
talentlens_graph = builder.compile()

from enum import Enum

from pydantic import BaseModel, Field, model_validator


class QuestionType(str, Enum):
    MCQ = "mcq"
    CODING = "coding"
    DEBUGGING = "debugging"
    SCENARIO = "scenario"
    SYSTEM_DESIGN = "system_design"
    SQL = "sql"


class Difficulty(str, Enum):
    EASY = "easy"
    MEDIUM = "medium"
    HARD = "hard"


class SkillAssessment(BaseModel):

    skill_name: str

    question_types: list[QuestionType]

    difficulty: Difficulty

    question_count: int = Field(gt=0, le=20)

    weightage: float = Field(gt=0, le=100)


class AssessmentPlan(BaseModel):

    total_questions: int = Field(gt=0)

    duration_minutes: int = Field(gt=0)

    skill_assessments: list[SkillAssessment]

    @model_validator(mode="after")
    def validate_plan(self):

        total_weight = sum(
            item.weightage
            for item in self.skill_assessments
        )

        if abs(total_weight - 100) > 0.01:
            raise ValueError(
                "Total skill weightage must equal 100%"
            )

        actual_questions = sum(
            item.question_count
            for item in self.skill_assessments
        )

        if actual_questions != self.total_questions:
            raise ValueError(
                "Total questions must match "
                "the sum of skill question counts"
            )

        return self

class AssessmentQuestion(BaseModel):
    question_id: str
    skill_name: str
    question_text: str
    question_type: QuestionType
    difficulty: Difficulty
    max_marks: float = Field(gt=0)
    time_limit_minutes: int = Field(gt=0)
    evaluation_rubric: list[str]
    expected_answer: str

class Assessment(BaseModel):

    assessment_id: str

    job_title: str

    assessment_plan: AssessmentPlan

    questions: list[AssessmentQuestion] = Field(min_length=1)

class GeneratedQuestionSet(BaseModel):

    questions: list[AssessmentQuestion] = Field(min_length=1)


class ReviewIssue(BaseModel):
    issue_type: str
    question_id: str | None = None
    description: str
    suggested_fix: str


class AssessmentReview(BaseModel):
    is_valid: bool
    overall_feedback: str
    issues: list[ReviewIssue] = Field(default_factory=list)
    revision_suggestions: list[str] = Field(default_factory=list)

import re
from pathlib import Path

from docx import Document

from app.schemas.assessment import Assessment


def extract_candidate_responses(
    file_path: str,
    assessment: Assessment
) -> list[dict]:

    document_path = Path(file_path)

    if not document_path.exists():
        raise FileNotFoundError(
            f"Candidate document not found: {file_path}"
        )

    if document_path.suffix.lower() != ".docx":
        raise ValueError("Only DOCX files are supported.")

    document = Document(str(document_path))

    answers = {}
    current_question = None
    collecting_answer = False

    for paragraph in document.paragraphs:

        text = paragraph.text.strip()

        # Detect question heading
        match = re.fullmatch(
            r"Question\s+(\d+)",
            text,
            re.IGNORECASE
        )

        if match:
            current_question = int(match.group(1))
            collecting_answer = False

            answers.setdefault(current_question, [])
            continue

        # Detect answer section
        if text.lower() == "your answer:":
            collecting_answer = True
            continue

        # Extract candidate's response
        if collecting_answer and current_question is not None:

            # Ignore blank lines and underscore placeholders
            if text and not re.fullmatch(r"_+", text):
                answers[current_question].append(text)

    # Map answers to original assessment questions
    candidate_responses = []

    for index, question in enumerate(
        assessment.questions,
        start=1
    ):

        answer_text = "\n".join(
            answers.get(index, [])
        ).strip()

        candidate_responses.append({
            "question_id": question.question_id,
            "skill_name": question.skill_name,
            "question_number": index,
            "candidate_answer": answer_text,
            "is_answered": bool(answer_text)
        })

    return candidate_responses
